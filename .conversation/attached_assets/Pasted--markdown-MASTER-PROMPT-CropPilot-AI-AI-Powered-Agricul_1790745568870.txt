### 10. Database Schema (Production SQL for PostgreSQL)

```sql
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email TEXT UNIQUE NOT NULL,
  password_hash TEXT NOT NULL,
  full_name TEXT NOT NULL,
  phone TEXT,
  preferred_language TEXT DEFAULT 'en' CHECK (preferred_language IN ('en', 'hi')),
  region_state TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE farms (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  name TEXT NOT NULL,
  state TEXT,
  district TEXT,
  soil_type TEXT,
  soil_ph DECIMAL(3,1),
  organic_matter TEXT,
  irrigation_method TEXT,
  farm_size_acres DECIMAL(10,2),
  primary_crops TEXT[],
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE diagnoses (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  farm_id UUID REFERENCES farms(id) ON DELETE SET NULL,
  crop_name TEXT NOT NULL,
  image_url TEXT,                    -- or store base64 / path
  symptoms_text TEXT,
  request_payload JSONB NOT NULL,
  ai_response JSONB NOT NULL,
  confidence_score INTEGER,
  language TEXT DEFAULT 'en',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE advice_reports (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  farm_id UUID REFERENCES farms(id) ON DELETE SET NULL,
  request_payload JSONB NOT NULL,
  ai_response JSONB NOT NULL,
  language TEXT DEFAULT 'en',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE chat_sessions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  title TEXT,
  language TEXT DEFAULT 'en',
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE chat_messages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  session_id UUID NOT NULL REFERENCES chat_sessions(id) ON DELETE CASCADE,
  role TEXT NOT NULL CHECK (role IN ('user', 'assistant')),
  content TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Indexes
CREATE INDEX idx_farms_user_id ON farms(user_id);
CREATE INDEX idx_diagnoses_user_id ON diagnoses(user_id);
CREATE INDEX idx_advice_reports_user_id ON advice_reports(user_id);
CREATE INDEX idx_chat_sessions_user_id ON chat_sessions(user_id);
CREATE INDEX idx_chat_messages_session_id ON chat_messages(session_id);
```

### 11. Row Level Security (RLS) / Data Isolation Rules

```sql
ALTER TABLE users ENABLE ROW LEVEL SECURITY;
ALTER TABLE farms ENABLE ROW LEVEL SECURITY;
ALTER TABLE diagnoses ENABLE ROW LEVEL SECURITY;
ALTER TABLE advice_reports ENABLE ROW LEVEL SECURITY;
ALTER TABLE chat_sessions ENABLE ROW LEVEL SECURITY;
ALTER TABLE chat_messages ENABLE ROW LEVEL SECURITY;

CREATE POLICY users_isolation ON users
  FOR ALL USING (id = current_setting('app.current_user_id')::UUID);

CREATE POLICY farms_isolation ON farms
  FOR ALL USING (user_id = current_setting('app.current_user_id')::UUID);

CREATE POLICY diagnoses_isolation ON diagnoses
  FOR ALL USING (user_id = current_setting('app.current_user_id')::UUID);

CREATE POLICY advice_reports_isolation ON advice_reports
  FOR ALL USING (user_id = current_setting('app.current_user_id')::UUID);

CREATE POLICY chat_sessions_isolation ON chat_sessions
  FOR ALL USING (user_id = current_setting('app.current_user_id')::UUID);

CREATE POLICY chat_messages_isolation ON chat_messages
  FOR ALL USING (
    session_id IN (
      SELECT id FROM chat_sessions 
      WHERE user_id = current_setting('app.current_user_id')::UUID
    )
  );
```

Backend must always run `SET app.current_user_id = '<authenticated_user_uuid>';` after JWT verification and before any database query.

### 13. Gemini SDK Setup & Server-Side Security

- Install `@google/genai`
- Initialize client only on server with `process.env.GEMINI_API_KEY`
- Never expose the key to the browser
- Support both `generateContent` (text) and multimodal (image + text)
- For images: accept multipart/form-data, convert to base64 or inline data for Gemini
- Force `responseMimeType: "application/json"` where structured output is required
- Implement retry + exponential backoff
- Rate-limit diagnosis and chat endpoints heavily
- Log only metadata (user_id, type, timestamp) — never full prompts containing PII or images
